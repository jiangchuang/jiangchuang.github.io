-- 测试索引大小

最佳实践：初始化大小写敏感，查询时动态设置

型号、颜色、容量 联合主键的例子

  1. 创建不可变的md5函数
   CREATE FUNCTION fn_md5_var(NAME VARCHAR) RETURN VARCHAR DETERMINISTIC AS
   BEGIN
     RETURN to_char(md5(lower(NAME));
   END;
  1. 增加虚拟列
   alter table tb_user_info ADD md5username as(fn_md5_lower(username));
  1. 在虚拟列上创建唯一索引
   -- 创建唯一索引前，请先清理重复数据(DM和dm已被识别为相同的用户名)
   create unique index idx_tb_user_info_md5username on tb_user_info(md5username);

-- 如果太大，可以截取md5前8位即可

select *,utl_raw.substr(md5name,1,8),utl_raw.substr(md5name,9,8) from fstest;
