查看绑定变量的值

程序中执行某条SQL性能有时跟管理工具效率相差较远，可以尝试查看程序中的绑定变量是否存在数据类型隐式转换、数据倾斜等问题

-- 开启监控绑定变量的参数
SF_SET_PARA_VALUE(1,'ENABLE_MONITOR_BP',1);

-- 根据其他条件搜索执行号 exec_id
SELECT exec_id,* FROM SYS."V$SQL_HISTORY"WHERE top_sql_text LIKE '%select%dual%'; -- 或者从sql日志中获取trx_id，根据trx_id过滤

-- 根据上一步中得到的执行号代入下方问号处
WITH SBH AS
 (SELECT EXEC_ID, SF_EXTRACT_BIND_DATA_NUM(BINDDATA, 1) AS ARG_NUM, BINDDATA
  FROM V$SQL_BINDDATA_HISTORY S
  WHERE EXEC_ID = ?)
SELECT SBH.EXEC_ID,
       SBH.ARG_NUM,
       LEVEL AS ARG_POS,
       SF_EXTRACT_BIND_DATA(BINDDATA, LEVEL, 1) AS ARG_TYPE,
       SF_EXTRACT_BIND_DATA(BINDDATA, LEVEL, 2) AS ARG_VAL
FROM SBH
CONNECT BY LEVEL <= ARG_NUM;

查看包调用方式

SELECT OBJECT_ID,
       OVERLOAD,
       PACKAGE_NAME,
       OBJECT_NAME,
       ARGUMENT_NAME,
       DATA_TYPE,
       DEFAULT_VALUE,
       IN_OUT,
       PACKAGE_NAME || '.' || OBJECT_NAME || '(' ||
       LISTAGG(ARGUMENT_NAME || '=>', ' , ')
               WITHIN GROUP (ORDER BY SEQUENCE)
               OVER (PARTITION BY OBJECT_ID,OBJECT_NAME,OVERLOAD) || DEFAULT_VALUE || ' );' AS CMD
FROM SYS.DBA_ARGUMENTS
WHERE OBJECT_NAME LIKE UPPER('%GET_DDL%')
  AND NOT (ARGUMENT_NAME = 'RET' AND IN_OUT = 'OUT');

查看每个表的空间使用情况

-- 通过页数和显示单位(可选参数MB/GB，默认KB为单位显示)
WITH FUNCTION GET_SIZE_BY_PAGES(NUM_OF_PAGE NUMBER, KB_MB_GB VARCHAR(2) DEFAULT 'KB') RETURN NUMBER(15, 2) AS
    V_SIZE_KB NUMBER(15, 2) := ROUND(PAGE() * NUM_OF_PAGE / 1024.00, 2);
BEGIN
    IF KB_MB_GB = 'MB' THEN
        RETURN V_SIZE_KB / 1024.00;
    ELSIF KB_MB_GB = 'GB' THEN
        RETURN V_SIZE_KB / 1024.00 / 1024.00;
    END IF;
    RETURN V_SIZE_KB;
END;
-- 查看每个表的空间占用情况和行数
SELECT T.TABLESPACE_NAME,
       T.OWNER,
       T.TABLE_NAME,
       GET_SIZE_BY_PAGES(TABLE_USED_PAGES(T.OWNER, T.TABLE_NAME), 'MB') AS TABLE_USED_SPACE_MB,
       GET_SIZE_BY_PAGES(TABLE_USED_PAGES(T.OWNER, T.TABLE_NAME), 'GB') AS TABLE_USED_SPACE_GB,
       TABLE_ROWCOUNT(T.OWNER, T.TABLE_NAME)                            AS TABLE_ROWCOUNT
FROM DBA_TABLES T
WHERE T.OWNER = USER -- 可更换USER为具体的模式名：如 T.OWNER = 'TEST'
ORDER BY TABLE_USED_SPACE_MB DESC, TABLE_ROWCOUNT DESC;

查看用户和模式对应

-- 查询用户模式对应
SELECT ur.name AS user_name, -- 用户名
       listagg(sch.name, ',') within GROUP(ORDER BY sch.id) AS schema_name -- 模式名
FROM sysobjects ur
JOIN sysobjects sch ON ur.id = sch.pid
WHERE ur."TYPE$" = 'UR'
      AND sch."TYPE$" = 'SCH'
GROUP BY ur.name
ORDER BY 1;

-- 查看表、字段、索引统计信息
WITH #STAT_INFO AS(
   SELECT SF_GET_SCHEMA_NAME_BY_ID(O.SCHID) AS SCH_NAME,
          NVL(I.TABLE_NAME, O.NAME) AS TABLE_NAME,
          NVL(C.NAME, I.COLUMN_NAME) AS COLUMN_NAME,
          I.INDEX_NAME,
          S.*
     FROM SYSSTATS S
     JOIN SYSOBJECTS O
       ON S.ID = O.ID
LEFT JOIN DBA_IND_COLUMNS I
       ON I.INDEX_NAME = O.NAME
      AND O."SUBTYPE$" = 'INDEX'
      AND SF_GET_SCHEMA_NAME_BY_ID(O.SCHID) = I.INDEX_OWNER
LEFT JOIN SYSCOLUMNS C
       ON S.ID = C.ID
      AND S.COLID = C.COLID
      AND S.T_FLAG='C')
SELECT SCH_NAME,TABLE_NAME,COLUMN_NAME,INDEX_NAME,T_TOTAL,N_SAMPLE,N_DISTINCT,N_NULL,BLEVEL,N_BUCKETS,COL_AVG_LEN,LAST_GATHERED
  FROM #STAT_INFO SI 
 WHERE SI.TABLE_NAME='EMPLOYEES'	-- 过滤表名
   AND SI.COLUMN_NAME LIKE '%LAST_NAME%';	-- 过滤列名

-- 可复制下方HOLD_LOCK_SESSION_KILL字段返回的kill命令结束长时间持有锁定的会话
SELECT H.SESS_ID AS HOLD_LOCK_SESSION_ID,
       CONCAT('SP_CLOSE_SESSION(', H.SESS_ID, ');') AS HOLD_LOCK_SESSION_KILL,
       H.TRX_ID AS HOLD_LOCK_TRX_ID,
       H.CLNT_HOST AS HOLD_LOCK_CLIENT_HOST,
       H.CLNT_IP AS HOLD_LOCK_CLIENT_IP,
       H.CLNT_VER AS HOLD_LOCK_CLNT_VER,

       W.SQL_TEXT AS WAIT_LOCK_SESSION_SQL,

查看等待事件历史

SELECT CAST(SESSADDR AS BIGINT) AS SESS_ID,

        FROM V$SQLTEXT S

内存类

查看数据库总内存占用

-- 查看总内存大小。也可以用来确定是数据缓存还是共享内存增长过快（增长前执行一次，增长后执行一次）
SELECT (SELECT SUM(n_pages * page_size) / 1024 / 1024
        FROM v$bufferpool) || 'MB' AS buffer_size,
       (SELECT SUM(total_size) / 1024 / 1024
        FROM v$mem_pool) || 'MB' AS mem_pool,
       (SELECT SUM(n_pages * page_size) / 1024 / 1024
        FROM v$bufferpool) + (SELECT SUM(total_size) / 1024 / 1024
                              FROM v$mem_pool) || 'MB' AS total_size
FROM dual;

查看BUFFER是否合理

-- 关注是否有频繁淘汰页面
--（1）如果 free 较大，说明该缓冲区空闲，可以适当的调整降低 buffer 缓冲 区参数值。
--（2）如果 free 项为 0，或者 N_DISCARD64 非零，表示该缓冲区经常淘汰， 说明对应的缓冲区参数太小，导致频繁淘汰，需要调整对应的缓冲区的参数
SELECT NAME, --缓冲区名称
       n_pages, --页数
       free, --空闲页数目
       n_discard64 --淘汰的页数
FROM v$bufferpool;

查看共享内存是否合理

--（1）目标外扩展次数 如果长期大于 0，说明长期从池外扩展，可能存在内存泄露，需要重点关注。
--（2）若使用到备份池，需要保持高度关注，此时系统非常危险。
--（3）内存池创建的线程号 creator 可以与 session 的 thrd_id 关联，查看对应的某个会话的内存使用情况，查看方法可以参考单个会话内存使用情况（链接）。
--（4）若 当前已分配大小 比 内存池初始大小 小，说明内存池非常空闲，可以减小对应的初始内存，避免浪费。
--（5）若 内存池总大小 比 目标大小 大，说明内存池不够，经常向池外申请，需要把对应参数调大。
  SELECT NAME AS 内存池名称,
         IS_SHARED AS 是否是共享的,
         IS_OVERFLOW AS 是否用到了备份池,
         round(ORG_SIZE/1024.0/1024.0,2) AS 内存池初始大小,
         round(RESERVED_SIZE/1024.0/1024.0,2) AS 当前已分配大小,
         round(TOTAL_SIZE/1024.0/1024.0,2) AS 内存池总大小,
         TARGET_SIZE/1024.0/1024.0 AS 目标大小,
--         round(DATA_SIZE/1024.0/1024.0,2) AS 实际有效字节,
--         EXTEND_SIZE AS 每次扩展大小,
         N_EXTEND_NORMAL AS 目标内扩展次数,
         N_EXTEND_EXCLUSIVE AS 目标外扩展次数
    FROM V$MEM_POOL
ORDER BY TOTAL_SIZE DESC;

自定义备份

create or replace procedure pro_db_backup_manual(p_type int default 0,
                                                 p_path varchar default '') as
  v_filename_tail       VARCHAR(50) := to_char(sysdate, 'yymmdd_hh24mi');
  v_backupset_name_full VARCHAR(50) := 'DB_BAK_FULL_' || v_filename_tail;
  v_backupset_name_inc  VARCHAR(50) := 'DB_BAK_INC_' || v_filename_tail;
  v_os_path_tail        VARCHAR(2) default '\';
  v_backup_cmd_full     varchar(1000);
  v_backup_cmd_inc      varchar(1000);
begin
  -- 如果未指定备份路径，则使用dm.ini中的BAK_PATH参数作为默认值
  if p_path = '' then
    select para_value
      into p_path
      from v$dm_ini
     where para_name = 'BAK_PATH';
  end if;
  -- 如果是Linux路径格式，则目录路径尾巴使用'/'分隔符
  if instr(p_path, '/') > 0 then
    v_os_path_tail := '/';
  end if;
  -- 去掉用户指定的路径斜杠尾巴
  p_path := rtrim(rtrim(p_path, '/'), '\');

  v_backup_cmd_full := 'backup database backupset ''' || p_path ||
                       v_os_path_tail || '' || v_backupset_name_full || '''';

  v_backup_cmd_inc := 'backup database increment with backupdir ''' ||
                      p_path || ''' backupset ''' || p_path ||
                      v_os_path_tail || v_backupset_name_inc || '''';
  -- 执行全量备份命令
  if p_type = 0 THEN
    EXECUTE IMMEDIATE v_backup_cmd_full;
  else
    begin
      -- 执行增量备份命令
      EXECUTE IMMEDIATE v_backup_cmd_inc;
    exception
      when others THEN
        -- 如果增备出错，则在指定的路径下立即执行一次全量备份
        EXECUTE IMMEDIATE v_backup_cmd_full;
    end;
  end if;
end;
